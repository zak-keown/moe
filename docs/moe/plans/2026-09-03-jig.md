# Jig — Tier 1 Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `subagent-driven-development` (recommended) or `executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship `packages/jig` as a working `moe-jig` CLI with five tier-1 commands (worktree create/remove/validate, plan init, spec init), two PreToolUse enforcement hooks, dispatcher integration, and npm publishing readiness.

**Architecture:** A new L0 TypeScript package (`packages/jig`) with zero workspace dependencies. The CLI uses `commander` for subcommand dispatch and shells out to `git` via `child_process`. Enforcement hooks ship with `packages/core/hooks/` and block raw commands in Claude Code sessions, redirecting to jig commands. The dispatcher (`bin/moe.js`) gains a `jig` namespace entry.

**Tech Stack:** TypeScript 5.9, commander, Node stdlib (child_process, fs, path), vitest.

**Spec:** `docs/moe/specs/2026-09-03-jig-design.md`

## Global Constraints

- Node >= 24, pnpm 11.23.0.
- No cross-package imports. `packages/jig` sits at L0 in the dependency graph.
- `tsconfig.json` extends `../../tsconfig.base.json` with empty `references: []`.
- Root `tsconfig.json` gains a `{ "path": "packages/jig" }` reference.
- `pnpm-workspace.yaml` already covers `packages/*` — no change needed.
- Every git operation uses `child_process.execFileSync` or `execFile` with argument arrays — never shell interpolation.
- Hooks use the polyglot `run-hook.cmd` wrapper for cross-platform compatibility.
- The publish workflow (`publish.yml`) adds `jig` to its package loop.

## Open Decisions

*None. All decisions were resolved during the design phase.*

## Out of Scope

- Tier 2 commands (review stamp, commit review-fix, iterations init). Separate plan after tier 1 ships.
- Tier 3 commands (context init, adr create, progress update). Same.
- Existing script migration (sdd-workspace, review-package, etc.). Separate plan.
- Skill updates to reference jig commands. A follow-up change after the CLI ships.
- Cleaning up existing worktrees in `.claude/worktrees/` or `.worktrees/`. Manual operation.

```typescript
import { dirname, resolve } from 'node:path'

export function git(...args: string[]): string {
  return execFileSync('git', args, { encoding: 'utf-8' }).trim()
}

export function today(): string {
  const d = new Date()
  const y = d.getFullYear()
  const m = String(d.getMonth() + 1).padStart(2, '0')
  const day = String(d.getDate()).padStart(2, '0')
  return `${y}-${m}-${day}`
}

export function resolvePrimaryRoot(): string {
  const commonDir = git('rev-parse', '--git-common-dir')
  const resolved = resolve(commonDir, '..')
  return git('-C', resolved, 'rev-parse', '--show-toplevel')
}
```

```typescript
import { execFileSync } from 'node:child_process'
import { existsSync, mkdirSync, readFileSync, writeFileSync, appendFileSync } from 'node:fs'
import { join, resolve, isAbsolute } from 'node:path'

function gitIn(cwd: string, ...args: string[]): string {
  return execFileSync('git', args, { cwd, encoding: 'utf-8' }).trim()
}

function primaryRoot(cwd: string): string {
  const commonDir = gitIn(cwd, 'rev-parse', '--git-common-dir')
  const resolved = resolve(cwd, commonDir, '..')
  return gitIn(resolved, 'rev-parse', '--show-toplevel')
}

function defaultBranch(cwd: string): string {
  try {
    const ref = gitIn(cwd, 'symbolic-ref', 'refs/remotes/origin/HEAD')
    return ref.replace('refs/remotes/origin/', '')
  } catch {
    // No remote — check for main, then master, then whatever HEAD is
    try {
      gitIn(cwd, 'rev-parse', '--verify', 'refs/heads/main')
      return 'main'
    } catch {
      try {
        gitIn(cwd, 'rev-parse', '--verify', 'refs/heads/master')
        return 'master'
      } catch {
        return gitIn(cwd, 'branch', '--show-current')
      }
    }
  }
}

function ensureGitignored(root: string): void {
  const gitignorePath = join(root, '.gitignore')
  const entry = '.moe/worktrees/'

  if (existsSync(gitignorePath)) {
    const content = readFileSync(gitignorePath, 'utf-8')
    if (content.includes(entry)) return
    appendFileSync(gitignorePath, `\n${entry}\n`)
  } else {
    writeFileSync(gitignorePath, `${entry}\n`)
  }
}

export interface WorktreeCreateOpts {
  base?: string
  cwd?: string
}

export function worktreeCreate(branch: string, opts: WorktreeCreateOpts = {}): string {
  const cwd = opts.cwd ?? process.cwd()
  const root = primaryRoot(cwd)
  const worktreeDir = join(root, '.moe', 'worktrees')

  mkdirSync(worktreeDir, { recursive: true })
  ensureGitignored(root)

  const baseBranch = opts.base ?? defaultBranch(root)
  const baseSha = gitIn(root, 'rev-parse', `${baseBranch}^{commit}`)

  const wtPath = join(worktreeDir, branch)
  gitIn(root, 'worktree', 'add', wtPath, '-b', branch, baseSha)

  // Verify lineage
  const mergeBase = gitIn(wtPath, 'merge-base', 'HEAD', baseSha)
  if (mergeBase !== baseSha) {
    throw new Error(
      `Lineage verification failed: merge-base ${mergeBase} !== base ${baseSha}`
    )
  }

  return wtPath
}

export function worktreeRemove(pathOrBranch: string, opts: { cwd?: string } = {}): void {
  const cwd = opts.cwd ?? process.cwd()
  const root = primaryRoot(cwd)
  const worktreeDir = join(root, '.moe', 'worktrees')

  let wtPath: string
  if (isAbsolute(pathOrBranch)) {
    wtPath = pathOrBranch
  } else if (existsSync(join(worktreeDir, pathOrBranch))) {
    wtPath = join(worktreeDir, pathOrBranch)
  } else {
    throw new Error(
      `Worktree "${pathOrBranch}" not found in ${worktreeDir}. ` +
      `jig only removes worktrees it created in .moe/worktrees/.`
    )
  }

  if (!wtPath.startsWith(worktreeDir)) {
    throw new Error(
      `Refusing to remove "${wtPath}" — it is outside .moe/worktrees/. ` +
      `Worktrees in .claude/worktrees/ belong to the harness; remove them there.`
    )
  }

  gitIn(root, 'worktree', 'remove', '--force', wtPath)
}

export interface ValidateResult {
  valid: boolean
  diagnostics: string[]
}

export function worktreeValidate(paths: string[]): ValidateResult {
  const diagnostics: string[] = []

  // Condition 1: each path is a linked worktree (not main, not submodule)
  for (const p of paths) {
    try {
      const gitDir = resolve(p, gitIn(p, 'rev-parse', '--git-dir'))
      const commonDir = resolve(p, gitIn(p, 'rev-parse', '--git-common-dir'))
      if (gitDir === commonDir) {
        diagnostics.push(`${p}: is the main checkout, not a linked worktree`)
        continue
      }
      const superproject = execFileSync(
        'git', ['-C', p, 'rev-parse', '--show-superproject-working-tree'],
        { encoding: 'utf-8' }
      ).trim()
      if (superproject) {
        diagnostics.push(`${p}: is a submodule, not a linked worktree`)
      }
    } catch {
      diagnostics.push(`${p}: not a valid git directory`)
    }
  }

  // Condition 2: paths are pairwise unique
  const resolved = paths.map((p) => resolve(p))
  const uniquePaths = new Set(resolved)
  if (uniquePaths.size !== resolved.length) {
    diagnostics.push('paths are not pairwise unique')
  }

  // Condition 3: git directories are pairwise unique
  const gitDirs: string[] = []
  for (const p of paths) {
    try {
      gitDirs.push(resolve(p, gitIn(p, 'rev-parse', '--git-dir')))
    } catch { /* already reported above */ }
  }
  const uniqueGitDirs = new Set(gitDirs)
  if (uniqueGitDirs.size !== gitDirs.length) {
    diagnostics.push('git directories are not pairwise unique')
  }

  // Condition 4: no path is a prefix of another
  for (let i = 0; i < resolved.length; i++) {
    for (let j = i + 1; j < resolved.length; j++) {
      if (resolved[i]!.startsWith(resolved[j]! + '/') ||
          resolved[j]!.startsWith(resolved[i]! + '/')) {
        diagnostics.push(`path prefix overlap: ${paths[i]} and ${paths[j]}`)
      }
    }
  }

  return { valid: diagnostics.length === 0, diagnostics }
}
```

```typescript
export function planInit(name: string, opts: { cwd?: string } = {}): string {
  const root = opts.cwd ?? process.cwd()
  const dir = join(root, 'docs', 'moe', 'plans')
  mkdirSync(dir, { recursive: true })

  const filename = `${today()}-${name}.md`
  const filepath = join(dir, filename)

  if (existsSync(filepath)) {
    throw new Error(`${filepath} already exists — refusing to overwrite`)
  }

  writeFileSync(filepath, PLAN_SKELETON(name))
  return filepath
}
```

```typescript
const HOOKS_DIR = join(import.meta.dirname, '..', '..', 'core', 'hooks')

function runHook(script: string, input: object): { code: number, stderr: string } {
  try {
    execFileSync('bash', [join(HOOKS_DIR, script)], {
      input: JSON.stringify(input),
      encoding: 'utf-8',
      env: { ...process.env, MOE_JIG_RAW_WORKTREE: '' },

    return { code: 0, stderr: '' }
```

- [ ] **Step 2: Create `packages/core/hooks/jig-worktree-guard`**

```bash
#!/bin/bash
# jig-worktree-guard: PreToolUse hook that blocks raw `git worktree add`.
# Redirects to `moe jig worktree create` which handles placement, gitignore,
# and lineage verification.
#
# Escape hatch: MOE_JIG_RAW_WORKTREE=1 disables this hook.
set -u

if [ -n "${MOE_JIG_RAW_WORKTREE:-}" ]; then
  exit 0
fi

EVENT="$(cat 2>/dev/null || true)"
if [ -z "$EVENT" ]; then
  exit 0
fi

# Only act on Bash tool calls
TOOL_NAME="$(printf '%s' "$EVENT" | grep -o '"tool_name"\s*:\s*"[^"]*"' | head -1 | sed 's/.*"\([^"]*\)"$/\1/')"
if [ "$TOOL_NAME" != "Bash" ]; then
  exit 0
fi

COMMAND="$(printf '%s' "$EVENT" | grep -o '"command"\s*:\s*"[^"]*"' | head -1 | sed 's/.*"\([^"]*\)"$/\1/')"

if printf '%s' "$COMMAND" | grep -qE 'git\s+worktree\s+add\b'; then
  cat >&2 <<'MSG'
BLOCKED: raw `git worktree add` is not allowed.
Use `moe jig worktree create <branch> [--base <ref>]` instead.
This ensures worktrees land in .moe/worktrees/, are gitignored,
and have verified lineage.
MSG
  exit 2
fi

exit 0
```

Make it executable: `chmod +x packages/core/hooks/jig-worktree-guard`

```bash
#!/bin/bash
# jig-review-format-guard: PreToolUse hook that validates commit messages
# containing `fix(review)` match the CR-### contract.
set -u

EVENT="$(cat 2>/dev/null || true)"
if [ -z "$EVENT" ]; then
  exit 0
fi

TOOL_NAME="$(printf '%s' "$EVENT" | grep -o '"tool_name"\s*:\s*"[^"]*"' | head -1 | sed 's/.*"\([^"]*\)"$/\1/')"
if [ "$TOOL_NAME" != "Bash" ]; then
  exit 0
fi

COMMAND="$(printf '%s' "$EVENT" | grep -o '"command"\s*:\s*"[^"]*"' | head -1 | sed 's/.*"\([^"]*\)"$/\1/')"

# Only check commits with fix(review) in the message
if ! printf '%s' "$COMMAND" | grep -qE 'git\s+commit\b'; then
  exit 0
fi
if ! printf '%s' "$COMMAND" | grep -qF 'fix(review)'; then
  exit 0
fi

# Validate format: fix(review): CR-### — <title>
if ! printf '%s' "$COMMAND" | grep -qE 'fix\(review\):\s*CR-[0-9]{3}\s*(—|--)\s*.+'; then
  cat >&2 <<'MSG'
BLOCKED: review-fix commit message does not match the required format.
Expected: fix(review): CR-### — <title>
Example:  fix(review): CR-012 — handle nil pointer in parser

Use `moe jig commit review-fix <CR-ID> <title>` to format it correctly.
MSG
  exit 2
fi

exit 0
```
