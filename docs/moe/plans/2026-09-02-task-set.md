# Intra-Plan Task DAG (`task-set`) Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `subagent-driven-development` (recommended) or `executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add a `task-set` CLI that deterministically computes intra-plan task waves, ready-sets, and structural validation from explicit `depends_on:` edges in plan markdown.

**Architecture:** A single extensionless Node script at `packages/core/hooks/task-set`, sibling to `plan-set`. Parses plan markdown to extract task metadata (number, `depends_on:`, `Files:`, `Consumes:`, `Produces:`, checkbox state), builds a dependency DAG, and exposes three verbs: `check` (validate), `waves` (compute parallel groups), `next` (ready-set from checkbox state). Node built-ins only, no dependencies.

**Tech Stack:** Node 24, Vitest

**Spec:** `docs/moe/specs/2026-09-02-task-set-design.md`

## Global Constraints

- Node built-ins only. The CLI ships as a plugin hook and cannot assume `pnpm install`.
- Plan markdown is the parser's input. The parser must tolerate code blocks, nested lists, and arbitrary prose without false positives on task-header detection.
- `depends_on:` is the machine-readable edge. `Consumes:`/`Produces:` remains human documentation. The two are not cross-validated.
- Source belongs under `packages/`; never hand-edit `plugins/`.
- Every task must keep `pnpm --filter @bubstack/moe-core test` and typecheck green before commit.

## Open Decisions

None. All design questions were resolved in the spec.

## Out of Scope

- Cross-manifest chaining (two manifests depending on each other). Separate problem.
- Parallel dispatch mechanics. `task-set waves` computes what *could* run in parallel; actually dispatching them remains gated on `parallel-execution-option`.
- Backfilling `depends_on:` into existing plans. Existing plans without `depends_on:` are valid (all tasks independent).
- Graph visualization.

```javascript
#!/usr/bin/env node
// task-set: the deterministic intra-plan task scheduler.
//
// Companion to plan-set (which sequences plans in a project). This CLI
// sequences tasks within a single plan: structural validation, wave
// grouping, and ready-set computation from checkbox state.
//
// Verbs:
//   check <plan.md>    Validate: task numbers unique, all depends_on resolve,
//                      no cycle (Kahn), every task has Files/Consumes/Produces.
//   waves <plan.md>    Print wave assignment (maximal parallel groups).
//   next  <plan.md>    Print ready set (deps complete, own steps not all done).
//
// Same constraints as plan-set: Node built-ins only, ships as a plugin hook.

import { existsSync, readFileSync } from "node:fs";
import { resolve } from "node:path";

const USAGE = `task-set: compute the intra-plan task DAG.

Usage:
  task-set check <plan.md>
  task-set waves <plan.md>
  task-set next  <plan.md>`;

function die(msg, code = 2) {
  process.stderr.write(`task-set: ${msg}\n`);
  process.exit(code);
}

// ---------------------------------------------------------------------------
// Plan parser.
//
// Extracts tasks from plan markdown. A task starts at a heading matching
// `### Task N:` and ends at the next such heading or EOF. Inside a task,
// the parser looks for:
//   - `**depends_on:** [1, 3]` — bracket-delimited integer list
//   - `**Files:**` block — lines starting with `- Create:`, `- Modify:`, `- Test:`
//   - `- Consumes:` and `- Produces:` lines (presence, not content)
//   - `- [ ]` and `- [x]` checkboxes (step completion state)
//
// The parser must skip fenced code blocks to avoid false positives on
// task-heading patterns inside code examples.

function parsePlan(text) {
  const lines = text.split(/\r?\n/);
  const tasks = [];
  let current = null;
  let inFence = false;

  const pushCurrent = () => {
    if (current !== null) tasks.push(current);
    current = null;
  };

  for (const line of lines) {
    // Track fenced code blocks to avoid false positives.
    if (/^(`{3,}|~{3,})/.test(line)) {
      inFence = !inFence;
      continue;
    }
    if (inFence) continue;

    // Task heading: ### Task N: Title
    const taskMatch = /^###\s+Task\s+(\d+)\s*:\s*(.*)$/.exec(line);
    if (taskMatch) {
      pushCurrent();
      current = {
        num: parseInt(taskMatch[1], 10),
        title: taskMatch[2].trim(),
        dependsOn: [],
        files: [],
        hasConsumes: false,
        hasProduces: false,
        steps: [],
      };
      continue;
    }

    if (!current) continue;

    // depends_on: [1, 3] or []
    const depMatch = /^\*\*depends_on:\*\*\s*\[([^\]]*)\]/.exec(line);
    if (depMatch) {
      const inner = depMatch[1].trim();
      if (inner !== "") {
        current.dependsOn = inner.split(",").map((s) => {
          const n = parseInt(s.trim(), 10);
          if (Number.isNaN(n)) die(`task ${current.num}: depends_on contains non-integer "${s.trim()}"`);
          return n;
        });
      }
      continue;
    }

    // Files: entries
    const fileMatch = /^-\s+(?:Create|Modify|Test):\s*`([^`]+)`/.exec(line);
    if (fileMatch) {
      current.files.push(fileMatch[1]);
      continue;
    }

    // Consumes / Produces presence
    if (/^-\s+Consumes:/.test(line)) {
      current.hasConsumes = true;
      continue;
    }
    if (/^-\s+Produces:/.test(line)) {
      current.hasProduces = true;
      continue;
    }

    // Checkbox steps
    const checkMatch = /^-\s+\[([ xX])\]/.exec(line);
    if (checkMatch) {
      current.steps.push({ checked: checkMatch[1] !== " " });
      continue;
    }
  }
  pushCurrent();
  return { tasks };
}

// ---------------------------------------------------------------------------
// Validation (shared by check and every other verb).

function validate(tasks) {
  const errors = [];
  const nums = new Map();

  for (const t of tasks) {
    if (nums.has(t.num)) {
      errors.push(`duplicate task number ${t.num}`);
    } else {
      nums.set(t.num, t);
    }
  }

  const known = new Set(nums.keys());

  for (const t of tasks) {
    if (t.files.length === 0) {
      errors.push(`task ${t.num} (${t.title}): missing "Files:" block`);
    }
    if (!t.hasConsumes) {
      errors.push(`task ${t.num} (${t.title}): missing "Consumes:" entry`);
    }
    if (!t.hasProduces) {
      errors.push(`task ${t.num} (${t.title}): missing "Produces:" entry`);
    }
    for (const d of t.dependsOn) {
      if (!known.has(d)) {
        errors.push(`task ${t.num}: depends_on ${d} — not a known task`);
      }
    }
  }

  // Cycle detection (Kahn).
  const indeg = new Map();
  const adj = new Map();
  for (const n of known) {
    indeg.set(n, 0);
    adj.set(n, []);
  }
  for (const t of tasks) {
    for (const d of t.dependsOn) {
      if (!known.has(d)) continue;
      indeg.set(t.num, (indeg.get(t.num) ?? 0) + 1);
      adj.get(d).push(t.num);
    }
  }
  const q = [];
  const remaining = new Map(indeg);
  for (const [n, deg] of remaining) if (deg === 0) q.push(n);
  let seen = 0;
  while (q.length) {
    const n = q.shift();
    seen++;
    for (const nb of adj.get(n) ?? []) {
      remaining.set(nb, remaining.get(nb) - 1);
      if (remaining.get(nb) === 0) q.push(nb);
    }
  }
  const inCycle = [...remaining.entries()].filter(([, d]) => d > 0).map(([n]) => n);
  if (inCycle.length > 0) {
    errors.push(`cycle detected among: ${inCycle.sort((a, b) => a - b).join(", ")}`);
  }

  return { errors };
}

// ---------------------------------------------------------------------------
// Verbs.

function verbCheck(planPath) {
  const text = readFileSync(planPath, "utf8");
  const { tasks } = parsePlan(text);

  if (tasks.length === 0) die(`${planPath}: no tasks found`);

  const { errors } = validate(tasks);
  if (errors.length > 0) {
    for (const e of errors) process.stderr.write(`task-set: ${e}\n`);
    process.exit(1);
  }
  process.stdout.write(`task-set: ${planPath} ok — ${tasks.length} tasks\n`);
}

// ---------------------------------------------------------------------------
// Main.

function main() {
  const argv = process.argv.slice(2);
  if (argv.length === 0 || argv[0] === "--help" || argv[0] === "-h") {
    process.stdout.write(USAGE + "\n");
    process.exit(argv.length === 0 ? 2 : 0);
  }
  const verb = argv[0];
  const planPath = argv[1];

  if (verb === "check") {
    if (!planPath) die("usage: task-set check <plan.md>");
    if (!existsSync(planPath)) die(`plan not found: ${planPath}`);
    verbCheck(resolve(planPath));
    return;
  }
  if (verb === "waves") {
    if (!planPath) die("usage: task-set waves <plan.md>");
    if (!existsSync(planPath)) die(`plan not found: ${planPath}`);
    die(`waves: not yet implemented`);
    return;
  }
  if (verb === "next") {
    if (!planPath) die("usage: task-set next <plan.md>");
    if (!existsSync(planPath)) die(`plan not found: ${planPath}`);
    die(`next: not yet implemented`);
    return;
  }
  die(`unknown verb "${verb}"\n${USAGE}`);
}

main();
```

```javascript
// ---------------------------------------------------------------------------
// Wave computation.
//
// A wave is a maximal set of tasks where:
//   1. No task depends on another in the same wave (no depends_on edge).
//   2. No two tasks share a file path (pairwise-disjoint Files blocks).
//
// Algorithm: topological order by Kahn's, then greedily assign each task to
// the earliest wave whose constraints it satisfies. The topological order
// guarantees a task's dependencies are assigned to earlier waves.

function computeWaves(tasks) {
  const byNum = new Map();
  for (const t of tasks) byNum.set(t.num, t);

  // Topological sort (Kahn's) to get a valid processing order.
  const indeg = new Map();
  const adj = new Map();
  for (const t of tasks) {
    indeg.set(t.num, 0);
    adj.set(t.num, []);
  }
  for (const t of tasks) {
    for (const d of t.dependsOn) {
      indeg.set(t.num, (indeg.get(t.num) ?? 0) + 1);
      adj.get(d).push(t.num);
    }
  }
  const q = [];
  for (const [n, deg] of indeg) if (deg === 0) q.push(n);
  // Stable tie-break: sort by task number so output is deterministic.
  q.sort((a, b) => a - b);
  const order = [];
  while (q.length) {
    const n = q.shift();
    order.push(n);
    const neighbors = (adj.get(n) ?? []).slice().sort((a, b) => a - b);
    for (const nb of neighbors) {
      indeg.set(nb, indeg.get(nb) - 1);
      if (indeg.get(nb) === 0) q.push(nb);
    }
    q.sort((a, b) => a - b);
  }

  // Assign each task in topological order to the earliest valid wave.
  // A task can join wave W if:
  //   - None of its depends_on targets are in wave W.
  //   - No task already in wave W shares a file path.
  const taskWave = new Map(); // task num -> wave index
  const waves = []; // waves[i] = [task nums]
  const waveFiles = []; // waveFiles[i] = Set of file paths

  for (const num of order) {
    const t = byNum.get(num);

    // Earliest wave this task could join: one past the latest wave
    // containing any of its dependencies.
    let earliest = 0;
    for (const d of t.dependsOn) {
      const dw = taskWave.get(d);
      if (dw !== undefined && dw + 1 > earliest) earliest = dw + 1;
    }

    // From earliest onward, find the first wave with no file overlap.
    let assigned = -1;
    for (let w = earliest; w < waves.length; w++) {
      const wFiles = waveFiles[w];
      const overlap = t.files.some((f) => wFiles.has(f));
      if (!overlap) {
        assigned = w;
        break;
      }
    }

    if (assigned === -1) {
      // No existing wave works; create a new one.
      assigned = waves.length;
      waves.push([]);
      waveFiles.push(new Set());
    }

    waves[assigned].push(num);
    taskWave.set(num, assigned);
    for (const f of t.files) waveFiles[assigned].add(f);
  }

  return waves;
}

function verbWaves(planPath) {

  if (errors.length > 0) {
    for (const e of errors) process.stderr.write(`task-set: ${e}\n`);
    process.exit(1);
  }
  const waves = computeWaves(tasks);
  for (let i = 0; i < waves.length; i++) {
    process.stdout.write(`wave ${i + 1}: ${waves[i].join(", ")}\n`);
  }
}

function verbNext(planPath) {
  const text = readFileSync(planPath, "utf8");
  const { tasks } = parsePlan(text);
  if (tasks.length === 0) die(`${planPath}: no tasks found`);
  const { errors } = validate(tasks);
  if (errors.length > 0) {
    for (const e of errors) process.stderr.write(`task-set: ${e}\n`);
    process.exit(1);
  }

  // A task is complete when every checkbox is checked.
  const complete = new Set();
  for (const t of tasks) {
    if (t.steps.length > 0 && t.steps.every((s) => s.checked)) {
      complete.add(t.num);
    }
  }

  // A task is ready when all deps are complete and it is not complete itself.
  const ready = [];
  for (const t of tasks) {
    if (complete.has(t.num)) continue;
    const allDepsComplete = t.dependsOn.every((d) => complete.has(d));
    if (allDepsComplete) ready.push(t.num);
  }

  for (const num of ready) process.stdout.write(num + "\n");
}
```
